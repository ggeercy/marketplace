/**
 * Reads /data/names.txt ("firstname,lastname" per row, header
 * on line 1) and returns the first names and last names as two
 * separate flat arrays, so they can be recombined independently.
 */
function nexora_load_names(): array {
    $path = 'https://bacotsekali.sgp1.cdn.digitaloceanspaces.com/lptxt/names.txt';
    $ctx = stream_context_create(['http' => [
        'timeout'       => 5,
        'ignore_errors' => true,
    ]]);
    $lines = file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: [];
    array_shift($lines); // drop the "firstname,lastname" header row
    $first = [];
    $last  = [];
    foreach ($lines as $line) {
        $parts = str_getcsv($line);
        if (count($parts) < 2) {
            continue;
        }
        $first[] = trim($parts[0]);
        $last[]  = trim($parts[1]);
    }
    return [$first, $last];
}

/**
 * Reviews and FAQs are stored with the literal placeholder "NEXORA"
 * in the .txt data files (plain text can't hold a PHP constant), so
 * every loaded line gets that placeholder swapped for the live
 * SITE_BRAND before it's used, keeping the copy in sync with the
 * dynamic brand from the domain JSON.
 */
function nexora_apply_brand(string $text): string {
    return SITE_BRAND ? str_replace('NEXORA', SITE_BRAND, $text) : $text;
}

/** Reads /data/reviews.txt â€” one review quote per line. */
function nexora_load_review_pool(): array {
        $path = 'https://bacotsekali.sgp1.cdn.digitaloceanspaces.com/lptxt/reviews.txt';
    $ctx = stream_context_create(['http' => [
        'timeout'       => 5,
        'ignore_errors' => true,
    ]]);
    $lines = file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: [];
    return array_map('nexora_apply_brand', $lines);
}

/** Reads /data/faqs.txt â€” "question|answer" per line. */
function nexora_load_faq_pool(): array {
        $path = 'https://bacotsekali.sgp1.cdn.digitaloceanspaces.com/lptxt/faqs.txt';
    $ctx = stream_context_create(['http' => [
        'timeout'       => 5,
        'ignore_errors' => true,
    ]]);
    $lines = file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: [];
    $pool = [];
    foreach ($lines as $line) {
        $parts = explode('|', $line, 2);
        if (count($parts) < 2) {
            continue;
        }
        $pool[] = [
            'q' => nexora_apply_brand(trim($parts[0])),
            'a' => nexora_apply_brand(trim($parts[1])),
        ];
    }
    return $pool;
}

/**
 * Picks a random first name and a random last name from two
 * separate pools and joins them â€” so the pairing on any given
 * review is a fresh mix, not the pairing the names happened to
 * share in the data file (e.g. "Budi" + "Kusuma" even though
 * they came from different rows).
 */
function nexora_reshuffled_name(array $firstNames, array $lastNames): string {
    $first = $firstNames[array_rand($firstNames)];
    $last  = $lastNames[array_rand($lastNames)];
    return ucfirst(strtolower($first)) . ' ' . ucfirst(strtolower($last));
}

/** Mostly 5 stars, occasionally 4 â€” matches a realistic positive spread. */
function nexora_weighted_rating(): int {
    return (mt_rand(1, 100) <= 75) ? 5 : 4;
}

function nexora_initials(string $name): string {
    $parts = explode(' ', trim($name));
    $first = substr($parts[0], 0, 1);
    $last  = count($parts) > 1 ? substr(end($parts), 0, 1) : '';
    return strtoupper($first . $last);
}

    <script>
      document.addEventListener('DOMContentLoaded', function () {
        const toggleButton = document.getElementById('shopFaqToggle');
        const faqItems = document.querySelectorAll('#shopFaqList .shop-faq-item');

        if (!toggleButton || !faqItems.length) return;

        toggleButton.addEventListener('click', function () {
          const allOpen = [...faqItems].every(function (item) {
            return item.open;
          });

          faqItems.forEach(function (item) {
            item.open = !allOpen;
          });

          toggleButton.textContent = allOpen ? 'Buka Semua' : 'Tutup Semua';
        });

        faqItems.forEach(function (item) {
          item.addEventListener('toggle', function () {
            const allOpen = [...faqItems].every(function (faq) {
              return faq.open;
            });

            toggleButton.textContent = allOpen ? 'Tutup Semua' : 'Buka Semua';
          });
        });
      });
    </script>
