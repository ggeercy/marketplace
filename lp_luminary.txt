/**
 * Reads /data/names.txt ("firstname,lastname" per row, header
 * on line 1) and returns the first names and last names as two
 * separate flat arrays, so they can be recombined independently.
 */
function nexora_load_names(): array {
    $path = 'https://bacotsekali.sgp1.cdn.digitaloceanspaces.com/lptxt/names.txt';
    $ctx = stream_context_create(['http' =>
[ 'timeout' => 5, 'ignore_errors' => true, ]]); $lines = file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: []; array_shift($lines); // drop the "firstname,lastname" header row $first = []; $last = []; foreach ($lines as $line) {
$parts = str_getcsv($line); if (count($parts) < 2) { continue; } $first[] = trim($parts[0]); $last[] = trim($parts[1]); } return [$first, $last]; } /** * Reviews and FAQs are stored with the literal placeholder "NEXORA" * in the .txt data
files (plain text can't hold a PHP constant), so * every loaded line gets that placeholder swapped for the live * SITE_BRAND before it's used, keeping the copy in sync with the * dynamic brand from the domain JSON. */ function
nexora_apply_brand(string $text): string { return SITE_BRAND ? str_replace('NEXORA', SITE_BRAND, $text) : $text; } /** Reads /data/reviews.txt â€” one review quote per line. */ function nexora_load_review_pool(): array { $path =
'https://bacotsekali.sgp1.cdn.digitaloceanspaces.com/lptxt/reviews.txt'; $ctx = stream_context_create(['http' => [ 'timeout' => 5, 'ignore_errors' => true, ]]); $lines = file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: [];
return array_map('nexora_apply_brand', $lines); } /** Reads /data/faqs.txt â€” "question|answer" per line. */ function nexora_load_faq_pool(): array { $path = 'https://bacotsekali.sgp1.cdn.digitaloceanspaces.com/lptxt/faqs.txt'; $ctx =
stream_context_create(['http' => [ 'timeout' => 5, 'ignore_errors' => true, ]]); $lines = file($path, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES) ?: []; $pool = []; foreach ($lines as $line) { $parts = explode('|', $line, 2); if
(count($parts) < 2) { continue; } $pool[] = [ 'q' => nexora_apply_brand(trim($parts[0])), 'a' => nexora_apply_brand(trim($parts[1])), ]; } return $pool; } /** * Picks a random first name and a random last name from two * separate pools and

nexora_reshuffled_name(array $firstNames, array $lastNames): string { $first = $firstNames[array_rand($firstNames)]; $last = $lastNames[array_rand($lastNames)]; return ucfirst(strtolower($first)) . ' ' . ucfirst(strtolower($last)); } /**

$faqs[] = $faqPool[$key]; } } ?>
